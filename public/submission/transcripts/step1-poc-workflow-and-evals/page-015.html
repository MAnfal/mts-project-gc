import { assignTier, type TierRules } from "./tier";
import type { CompanyProfile, LabelSet, RunRecord } from "./types";

/**
 * Eval v2 grading (code only — no model). Every key case is either a FLAG (should be found) or a TRAP
 * (looks relevant, should be left out). Documents in neither list that a run flags are "unlabeled" extras.
 * Exact facts (date, tier, owner) are checked by code against the key; judgment calls go to the judge panel.
 */

export interface Ratio {
  correct: number;
  total: number;
}
const ratio = (correct = 0, total = 0): Ratio => ({ correct, total });
const add = (a: Ratio, b: Ratio): Ratio => ({ correct: a.correct + b.correct, total: a.total + b.total });

export interface RunGrade {
  runId: string;
  counts: { tp: number; fn: number; trapHit: number; tn: number; unlabeled: number };
  /** kind:direct, kind:indirect, trap:<type>, docType:<type> → correct / total */
  slices: Record<string, Ratio>;
  checks: { date: Ratio; tier: Ratio; owner: Ratio };
  /** Per key case: did this run get it right (flag found / trap left out)? Feeds pass^k. */
  caseCorrect: Record<string, boolean>;
  missed: string[];
  trapHits: string[];
  unlabeled: string[];
  /** Baseline claims that pointed at no real Federal Register document. */
  madeUp: number;
  costUsd: number;
  ms: number;
}

export function trapsOf(key: LabelSet) {
  // v1 keys called traps "borderlineRejected".
  if (key.traps) return key.traps;
  return Object.fromEntries(Object.entries(key.borderlineRejected ?? {}).map(([n, t]) => [n, { trapType: "other" as const, reason: t.reason, confidence: t.confidence }]));
}

export function gradeRun(run: RunRecord, key: LabelSet, profile: CompanyProfile, rules: TierRules): RunGrade {
  const flagged = new Map((run.items ?? []).map((i) => [i.document_number, i]));
  const traps = trapsOf(key);
  const slices: Record<string, Ratio> = {};
  const bump = (k: string, ok: boolean) => (slices[k] = add(slices[k] ?? ratio(), ratio(ok ? 1 : 0, 1)));
  const checks = { date: ratio(), tier: ratio(), owner: ratio() };
  const caseCorrect: Record<string, boolean> = {};

  for (const n of key.relevant) {
    const d = key.details?.[n];
    const it = flagged.get(n);
    caseCorrect[n] = !!it;
    bump(`kind:${d?.kind ?? "direct"}`, !!it);
    if (d?.docType) bump(`docType:${d.docType}`, !!it);
    if (!it || !d) continue;
    if (d.deadline) {
      const ok = !!it.deadline && [d.deadline.date, ...(d.acceptDates ?? [])].includes(it.deadline.date);
      checks.date = add(checks.date, ratio(ok ? 1 : 0, 1));
    }
    if (d.docType) {
      const expected = assignTier({
        type: d.docType,
        deadlines: d.deadline ? [{ date: d.deadline.date, kind: d.deadline.kind, source: "fr_field", verified: true }] : [],
        asOf: run.config.asOf,
        rules,
      }).tier;
      checks.tier = add(checks.tier, ratio(it.tier === expected ? 1 : 0, 1));
    }
    const want = profile.areas[d.area]?.owner.email;
    if (want) checks.owner = add(checks.owner, ratio(it.owner.email === want ? 1 : 0, 1));
  }
  for (const [n, t] of Object.entries(traps)) {
    const ok = !flagged.has(n);
    caseCorrect[n] = ok;
    bump(`trap:${t.trapType}`, ok);
  }

  const keyed = new Set([...key.relevant, ...Object.keys(traps)]);
  const missed = key.relevant.filter((n) => !flagged.has(n));
  const trapHits = Object.keys(traps).filter((n) => flagged.has(n));
  const unlabeled = [...flagged.keys()].filter((n) => !keyed.has(n));
  return {
    runId: run.id,
    counts: { tp: key.relevant.length - missed.length, fn: missed.length, trapHit: trapHits.length, tn: Object.keys(traps).length - trapHits.length, unlabeled: unlabeled.length },
    slices,
    checks,
    caseCorrect,
    missed,
    trapHits,
    unlabeled,
    madeUp: (run.dropped ?? []).filter((d) => /does not exist|no Federal Register/i.test(d.why)).length,
    costUsd: run.totalCostUsd + (run.judge?.costUsd ?? 0) + (run.panel?.costUsd ?? 0),
    ms: run.totalMs,
  };
}

/** pass^k (τ-bench): share of cases a setup got right in every one of k runs (uses the first k runs). */
export function passK(perRun: Record<string, boolean>[], k: number): number {
  const runs = perRun.slice(0, k);
  if (runs.length < k || !runs.length) return NaN;
  const cases = Object.keys(runs[0]);
  if (!cases.length) return NaN;
  // pass^k estimated over all k-subsets would be fairer for k < runs; with k ≤ runs.length we average subsets.
  if (k === 1) return runs.reduce((a, r) => a + cases.filter((c) => r[c]).length, 0) / (runs.length * cases.length);
  return cases.filter((c) => runs.every((r) => r[c])).length / cases.length;
}

export type Verdict = "PASS" | "FAIL" | "UNKNOWN";
/** Judge-panel vote: a verdict needs `minAgree` matching votes; UNKNOWN is an abstention. */
export function majority(votes: Verdict[], minAgree: number): "PASS" | "FAIL" | "CONTESTED" {
  const pass = votes.filter((v) => v === "PASS").length;
  const fail = votes.filter((v) => v === "FAIL").length;
  if (pass >= minAgree && pass > fail) return "PASS";
  if (fail >= minAgree && fail > pass) return "FAIL";
  return "CONTESTED";
}

/** Sum many run grades (e.g. one setup across every eval set) into headline numbers. */
export function aggregate(grades: RunGrade[]) {
  const counts = { tp: 0, fn: 0, trapHit: 0, tn: 0, unlabeled: 0 };
  const slices: Record<string, Ratio> = {};
  const checks = { date: ratio(), tier: ratio(), owner: ratio() };
  for (const g of grades) {
    for (const k of Object.keys(counts) as (keyof typeof counts)[]) counts[k] += g.counts[k];
    for (const [k, v] of Object.entries(g.slices)) slices[k] = add(slices[k] ?? ratio(), v);
    for (const k of ["date", "tier", "owner"] as const) checks[k] = add(checks[k], g.checks[k]);
  }
  const div = (a: number, b: number) => (b ? a / b : null);
  return {
    runs: grades.length,
    counts,
    slices,
    checks,
    recall: div(counts.tp, counts.tp + counts.fn),
    precision: div(counts.tp, counts.tp + counts.trapHit + counts.unlabeled),
    trapAvoidance: div(counts.tn, counts.tn + counts.trapHit),
    madeUp: grades.reduce((a, g) => a + g.madeUp, 0),
    costUsd: grades.reduce((a, g) => a + g.costUsd, 0),
    msMean: div(grades.reduce((a, g) => a + g.ms, 0), grades.length),
  };
}

import crypto from "node:crypto";
import { generateText, Output } from "ai";
import { z } from "zod";
import { judges, loadKey, prompts, render } from "./config";
import { gw, gatewayCost } from "./gateway";

/**
 * Judge panel (config/judges.yaml): each judge model grades every action item on every criterion in ONE call,
 * and a verdict needs `minAgree` matching votes. Verdicts are stored on the run with a hash of the panel config
 * and prompt, so editing the panel re-judges old runs without re-running the pipeline.
 */
export function panelHash(): string {
  const j = judges();
  return crypto.createHash("sha256").update(JSON.stringify(j)).update(prompts().panelJudge).digest("hex").slice(0, 10);
}

const VoteSchema = z.object({
  votes: z.array(z.object({ document_number: z.string(), criterion: z.string(), verdict: z.enum(["PASS", "FAIL", "UNKNOWN"]), why: z.string() })),
});

export async function judgePanel(run: RunRecord): Promise<PanelResult> {
  const cfg = judges();
  const key = loadKey(run.config.profileId, run.config.windowStart, run.config.windowEnd);
  const items = (run.items ?? []).map((i) => ({
    document_number: i.document_number,
    title: i.title,
    owner: i.owner.name,
    reason: i.reason,
    action: i.action,
    ...(key?.details?.[i.document_number]?.rubric ? { rubric: key.details[i.document_number].rubric } : {}),
  }));
  const hash = panelHash();
  if (!items.length) return { panelHash: hash, judges: [], verdicts: [], costUsd: 0 };

  const applies = (c: (typeof cfg.criteria)[number], it: (typeof items)[number]) => !c.needsRubric || "rubric" in it;
  const criteriaText = cfg.criteria.map((c) => `- ${c.id}: ${c.question}${c.needsRubric ? " (only for items that have a rubric)" : ""}`).join("\n");
  const prompt = render(prompts().panelJudge, { criteria: criteriaText, items: JSON.stringify(items, null, 1) });

  const results = await Promise.all(
    cfg.judges.map(async (j) => {
      try {
        const r = await generateText({ model: gw(j.model), output: Output.object({ schema: VoteSchema }), prompt, maxRetries: 2 });
        const costUsd = r.steps.reduce((a, s) => a + gatewayCost(s.providerMetadata), 0) || gatewayCost(r.providerMetadata);
        return { model: j.model, votes: r.output.votes as PanelVote[], costUsd };
      } catch (e) {
        return { model: j.model, votes: [] as PanelVote[], costUsd: 0, error: (e as Error).message.slice(0, 300) };
      }
    }),
  );

  const verdicts: PanelResult["verdicts"] = [];
  for (const it of items)
    for (const c of cfg.criteria) {
      if (!applies(c, it)) continue;
      const votes = results.map((r) => r.votes.find((v) => v.document_number === it.document_number && v.criterion === c.id)?.verdict ?? ("MISSING" as const));
      // A missing vote (judge errored or skipped the item) is an abstention, like UNKNOWN.
      const verdict = majority(votes.map((v) => (v === "MISSING" ? "UNKNOWN" : v)), cfg.minAgree);
      verdicts.push({ document_number: it.document_number, criterion: c.id, verdict, votes });
    }
  return { panelHash: hash, judges: results, verdicts, costUsd: results.reduce((a, r) => a + r.costUsd, 0) };
}

/** Panel summary for a set of runs: pass rate per criterion (over decided verdicts), contested share, judge agreement. */
export function panelSummary(runs: RunRecord[]) {
  const vs = runs.flatMap((r) => r.panel?.verdicts ?? []);
  const byCrit: Record<string, { pass: number; fail: number; contested: number }> = {};
  let unanimous = 0;
  for (const v of vs) {
    const b = (byCrit[v.criterion] ??= { pass: 0, fail: 0, contested: 0 });
    if (v.verdict === "PASS") b.pass++;
    else if (v.verdict === "FAIL") b.fail++;
    else b.contested++;
    const cast = v.votes.filter((x) => x === "PASS" || x === "FAIL");
    if (cast.length === v.votes.length && new Set(cast).size === 1) unanimous++;
  }
  const decided = vs.filter((v) => v.verdict !== "CONTESTED");
  return {
    verdicts: vs.length,
    passRate: decided.length ? decided.filter((v) => v.verdict === "PASS").length / decided.length : null,
    contested: vs.length ? (vs.length - decided.length) / vs.length : null,
    unanimous: vs.length ? unanimous / vs.length : null,
    byCriterion: byCrit,
  };
}
