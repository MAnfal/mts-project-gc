import fs from "node:fs";
import path from "node:path";
import YAML from "yaml";
import { DATA } from "./paths";

/**
 * Improvement iterations (data/iterations/<NN-slug>/): the story (hypothesis → change → result → decision),
 * the measured results, the evidence that motivated the change (error-analysis.json: code graders, trace checks
 * and judge verdicts on dev weeks), and the exact diffs (system.diff, config.diff).
 */
export interface IterationMetrics {
  setup: string;
  label: string;
  runs: number;
  recall: number | null;
  precision: number | null;
  trapAvoidance: number | null;
  datesRight: number | null;
  judgePass: number | null;
  consistency: number | null;
  score: number | null;
  perRun: Record<string, number>;
  devCasesRight: string | null;
  testCasesRight: string | null;
  costPerRunUsd: number;
  secondsPerRun: number | null;
}

export interface Iteration {
  id: string;
  title: string;
  date: string;
  suite: string;
  setup: string;
  comparedTo?: string;
  hypothesis: string;
  change: string;
  codeChanges?: string;
  evalChanges?: string;
  decision: string;
  notes?: string;
  regraded?: string;
  results: { candidate: IterationMetrics; comparedTo?: IterationMetrics };
  code?: { commit: string; uncommittedSrcChanges: number };
  systemChanges?: string[];
  configChangesSincePrevious?: { previous: string | null; files: string[] };
  systemDiff?: string;
  configDiff?: string;
  evidence?: { runs: number; excludedTestRuns?: number; bySignal: Record<string, number>; buckets: [string, number][]; rootCauses?: { cause: string; whose: string; failures: number; proposedChange: string }[] };
}

const read = (p: string) => (fs.existsSync(p) ? fs.readFileSync(p, "utf8") : undefined);

export function listIterations(): Iteration[] {
  const dir = path.join(DATA(), "iterations");
  if (!fs.existsSync(dir)) return [];
  return fs
    .readdirSync(dir)
    .filter((d) => fs.existsSync(path.join(dir, d, "iteration.yaml")))
    .sort()
    .map((d) => {
      const it = YAML.parse(fs.readFileSync(path.join(dir, d, "iteration.yaml"), "utf8")) as Iteration;
      const ea = read(path.join(dir, d, "error-analysis.json"));
      const e = ea ? JSON.parse(ea) : undefined;
      return {
        ...it,
        date: String(it.date),
        systemDiff: read(path.join(dir, d, "system.diff")),
        configDiff: read(path.join(dir, d, "config.diff")),
        evidence: e ? { runs: e.runs, excludedTestRuns: e.excludedTestRuns, bySignal: e.bySignal, buckets: e.buckets, rootCauses: e.llm?.rootCauses } : undefined,
      };
    });
}

const pct = (x: number | null | undefined) => (x === null || x === undefined ? "—" : `${(x * 100).toFixed(1)}%`);
const delta = (a: number | null | undefined, b: number | null | undefined) => {
  if (a === null || a === undefined || b === null || b === undefined) return null;
  const d = (a - b) * 100;
  return <span className={d >= 0 ? "text-[var(--status-good)]" : "text-destructive"}>{d >= 0 ? "+" : ""}{d.toFixed(1)}</span>;
};

/** A unified diff with +/− lines coloured (plain <pre>, no library needed). */
function Diff({ text }: { text: string }) {
  return (
    <pre className="text-xs overflow-x-auto rounded-md border bg-muted/30 p-3 max-h-96">
      {text.split("\n").map((l, i) => (
        <div key={i} className={l.startsWith("+") && !l.startsWith("+++") ? "text-[var(--status-good)]" : l.startsWith("-") && !l.startsWith("---") ? "text-destructive" : l.startsWith("@@") ? "text-muted-foreground" : ""}>
          {l || " "}
        </div>
      ))}
    </pre>
  );
}

// Recall / precision per split (dev vs held-out test) for each setup in a suite.
//   npx tsx scripts/split-report.mts <arenaId>
import { arenaRuns } from "../src/lib/arena";
import { evalsets, getCompany, loadKey, pipelineCfg } from "../src/lib/config";
import { aggregate, gradeRun } from "../src/lib/grade";
const runs = (await arenaRuns(process.argv[2])).filter((r) => r.status === "done");
const split = new Map(evalsets().map((e) => [e.id, e.split]));
const cells = [...new Set(runs.map((r) => r.config.cell!))];
for (const c of cells)
  for (const s of ["dev", "test"]) {
    const g = runs.filter((r) => r.config.cell === c && split.get(r.config.evalset!) === s).map((r) => gradeRun(r, loadKey(r.config.profileId, r.config.windowStart, r.config.windowEnd)!, getCompany(r.config.profileId), pipelineCfg().tiering));
    const a = aggregate(g);
    const p = (x: number | null) => (x === null ? "—" : `${(x * 100).toFixed(1)}%`);
    console.log(`${c.padEnd(24)} ${s.padEnd(4)} runs ${String(g.length).padStart(2)}  recall ${p(a.recall)} (${a.counts.tp}/${a.counts.tp + a.counts.fn})  precision ${p(a.precision)}  trapHits ${a.counts.trapHit} extras ${a.counts.unlabeled}`);
  }
EOF
