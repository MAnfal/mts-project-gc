export function newRun(config: RunConfig): RunRecord {
  const id = `${new Date().toISOString().replace(/[:.]/g, "-")}_${config.mode}_${Math.random().toString(36).slice(2, 6)}`;
  return { id, createdAt: new Date().toISOString(), config, status: "running", stages: [], calls: [], totalCostUsd: 0, totalMs: 0 };
}

export async function execute(run: RunRecord): Promise<RunRecord> {
  const ctx = new RunCtx(run);
  const t0 = Date.now();
  try {
    await ctx.init();
    await saveRun(run);
    if (run.config.mode === "baseline") await baseline(ctx);
    else await pipeline(ctx);
    run.status = "done";
  } catch (e) {
    run.status = "error";
    run.error = (e as Error).message;
  }
  run.totalMs = Date.now() - t0;
  await saveRun(run);
  return run;
}

async function pipeline(ctx: RunCtx) {
  const cfg = ctx.run.config;
  const profile = getProfile(cfg.profileId);

  const docs = await scan(ctx);
  ctx.run.scanned = docs.length;
  if (cfg.webLeads) ctx.run.webLeads = await webLeads(ctx, profile, docs);

  const decisions = new Map<string, FilterDecision>(docs.map((d) => [d.document_number, { document_number: d.document_number, relevant: false }]));
  let toLlm = docs;
  if (cfg.filter !== "llm") {
    const probs = await filterJev(ctx, profile, docs);
    for (const [n, p] of probs) decisions.get(n)!.jevProbability = p;
    toLlm = docs.filter((d) => (probs.get(d.document_number) ?? 0) >= cfg.jevThreshold);
    if (cfg.filter === "jev") for (const d of toLlm) decisions.get(d.document_number)!.relevant = true;
  }
  if (cfg.filter !== "jev") {
    const llm = await filterLlm(ctx, profile, toLlm);
    for (const [n, v] of llm) {
      const dec = decisions.get(n)!;
      dec.llm = v;
      dec.relevant = v.relevant;
    }
  }
  ctx.run.decisions = [...decisions.values()];

  const relevantDocs = docs.filter((d) => decisions.get(d.document_number)!.relevant);
  const { items, dropped } = await verifyAndTier(ctx, profile, relevantDocs, decisions);
  ctx.run.dropped = dropped;
  ctx.run.items = await writeActions(ctx, profile, items);
  ctx.run.markdown = renderMarkdown(profile, cfg, ctx.run.items);
}

export class BudgetExceeded extends Error {}

/** Per-run bookkeeping: every model call goes through here so cost, latency and I/O land in the transcript. */
export class RunCtx {
  private spentAtStart = 0;
  constructor(public run: RunRecord) {}

  async init() {
    // Excludes this run (not yet saved), so the guard below compares prior spend + this run's spend.
    this.spentAtStart = await spentUsd();
    if (this.spentAtStart >= BUDGET_USD) throw new BudgetExceeded(`Budget exhausted: $${this.spentAtStart.toFixed(2)} of $${BUDGET_USD}`);
  }

  private guard() {
    if (this.spentAtStart + this.run.totalCostUsd >= BUDGET_USD) {
      throw new BudgetExceeded(`Budget cap $${BUDGET_USD} reached mid-run`);
    }
  }

  private record(log: LlmCallLog) {
    this.run.calls.push(log);
    this.run.totalCostUsd += log.costUsd;
  }

  async stage<T>(name: string, fn: () => Promise<T>, counts?: (r: T) => { inCount?: number; outCount?: number; note?: string }) {
    const t0 = Date.now();
    const costBefore = this.run.totalCostUsd;
    const r = await fn();
    this.run.stages.push({ name, ms: Date.now() - t0, costUsd: this.run.totalCostUsd - costBefore, ...(counts?.(r) ?? {}) });
    await saveRun(this.run);
    return r;
  }

  /** Structured LLM call (optionally with tools). */
  async object<S extends z.ZodType>(stage: string, args: { model: string; system?: string; prompt: string; schema: S; tools?: ToolSet; logPrompt?: boolean }): Promise<z.infer<S>> {
    this.guard();
    const t0 = Date.now();
    try {
      const r = await generateText({
        model: gw(args.model),
        system: args.system,
        prompt: args.prompt,
        tools: args.tools,
        output: Output.object({ schema: args.schema }),
        maxRetries: 2,
      });
      const toolCalls = r.steps.flatMap((s) =>
        s.toolCalls.map((c) => ({
          name: c.toolName,
          input: c.input,
          output: s.toolResults.find((x) => x.toolCallId === c.toolCallId)?.output,
        })),
      );
      this.record({
        stage,
        model: args.model,
        ms: Date.now() - t0,
        costUsd: r.steps.reduce((a, s) => a + gatewayCost(s.providerMetadata), 0) || gatewayCost(r.providerMetadata),
        inputTokens: r.totalUsage.inputTokens,
        outputTokens: r.totalUsage.outputTokens,
        prompt: args.logPrompt === false ? undefined : args.prompt,
        output: r.output,
        toolCalls: toolCalls.length ? toolCalls : undefined,
      });
      return r.output as z.infer<S>;
    } catch (e) {
      this.record({ stage, model: args.model, ms: Date.now() - t0, costUsd: 0, prompt: args.prompt, error: (e as Error).message });
      throw e;
    }
  }

  /** Free-form tool loop (used for web leads). Returns text + every tool call. */
  async agent(stage: string, args: { model: string; system?: string; prompt: string; tools: ToolSet; maxSteps: number }) {
    this.guard();
    const t0 = Date.now();
    const { isStepCount } = await import("ai");
    const r = await generateText({ model: gw(args.model), system: args.system, prompt: args.prompt, tools: args.tools, stopWhen: isStepCount(args.maxSteps) });

    );
    this.record({
      stage,
      model: args.model,
      ms: Date.now() - t0,
      costUsd: r.steps.reduce((a, s) => a + gatewayCost(s.providerMetadata), 0),
      inputTokens: r.totalUsage.inputTokens,
      outputTokens: r.totalUsage.outputTokens,
      prompt: args.prompt,
      output: r.text,
      toolCalls,
    });
    return { text: r.text, toolCalls, sources: r.sources };
  }

  /** Jev (decision model): probability only, no text. Logged in aggregate to keep transcripts readable. */

    const a = r.answers.relevant as { probability?: number };
    return { p: a.probability ?? 0, cost, ms: Date.now() - t0 };
  }

/** Dates graded against the key: for found docs whose key has a deadline, does the shown date match exactly? */
export function datesAgainstKey(run: RunRecord, key: LabelSet) {
  const wrong: string[] = [];
  let checked = 0;
  for (const it of run.items ?? []) {
    const expected = key.details?.[it.document_number]?.deadline;
    if (!key.relevant.includes(it.document_number) || !expected) continue;
    checked++;
    if (it.deadline?.date !== expected.date) wrong.push(it.document_number);
  }

/**
 * Creates every run up front (so the page shows all cells immediately), then executes in waves of
 * CONCURRENCY. Each run snapshots spend at start and can't see siblings, so the budget is re-checked
 * between waves using the worst-case estimate of the next wave.

export async function startArena(args: { profileId: string; windowStart: string; windowEnd: string; asOf: string; cellKeys: string[]; repeat: number }) {
  const cells = args.cellKeys.map(cellByKey).filter((c): c is Cell => !!c);
  const est = estimate(cells, args.repeat);
  const spent = await spentUsd();
  if (spent + est > BUDGET_USD) throw new Error(`Estimated $${est.toFixed(2)} would exceed the budget ($${spent.toFixed(2)} of $${BUDGET_USD} spent)`);

  const arena = `arena-${new Date().toISOString().slice(0, 16).replace(/[:T]/g, "-")}-${Math.random().toString(36).slice(2, 5)}`;
  const runs: { run: RunRecord; cell: Cell }[] = [];
  for (let i = 0; i < args.repeat; i++) for (const cell of cells) runs.push({ run: newRun(cellConfig(cell, args, arena)), cell });
  await Promise.all(runs.map((r) => saveRun(r.run)));

  const go = async () => {
    const limit = pLimit(CONCURRENCY);
    for (let i = 0; i < runs.length; i += CONCURRENCY) {
      const wave = runs.slice(i, i + CONCURRENCY);
      const waveEst = wave.reduce((a, w) => a + w.cell.estUsd, 0);
      if ((await spentUsd()) + waveEst > BUDGET_USD) {
        for (const w of runs.slice(i)) {
          w.run.status = "error";
          w.run.error = "Skipped: budget cap would be exceeded";
          await saveRun(w.run);
        }
        return;
      }
      await Promise.all(
        wave.map((w) =>
          limit(async () => {
            await execute(w.run);
            if (w.run.status === "done") {
              w.run.judge = await judgeClarity(w.run);
              await saveRun(w.run);
            }
          }),
        ),
      );
    }
  };
  return { arena, runIds: runs.map((r) => r.run.id), estimateUsd: est, go };
}
